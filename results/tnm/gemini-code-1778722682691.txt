% =========================================================================
% TNM Model Simulation for Kovacs Memory Effect in Au-based Metallic Glass
% Based on parameters and Eq(1)-(3) from the provided literature
% =========================================================================

clc; clear; close all;

%% 1. 模型参数定义 (基于文献拟合结果)
H_star = 164e3;      % 激活焓 (J/mol) [cite: 57]
A = 6e-22;           % 指数前因子 (s) [cite: 57]
x = 0.6;             % 非线性参数 [cite: 57]
beta = 0.43;         % 扩展指数参数 (KWW) [cite: 57]
R = 8.314;           % 理想气体常数 (J/(mol*K))
T0 = 430;            % 平衡过冷液体的起始温度，即初始虚拟温度 (K) [cite: 49]

%% 2. 图1a: 单步等温退火 (Monotonic)
Ta_1a = 383; % 退火温度 [cite: 36, 271]
t1_1a = logspace(-1, 4, 1000); % 时间范围 10^-1 到 10^4 s
Tf_1a = zeros(size(t1_1a));
Tf_1a(1) = T0; 
S1_1a = 0; % 积分项初始化

for i = 2:length(t1_1a)
    dt = t1_1a(i) - t1_1a(i-1);
    % 公式3: 计算结构弛豫时间 [cite: 54, 55]
    tau = exp(log(A) + (x * H_star) / (R * Ta_1a) + ((1 - x) * H_star) / (R * Tf_1a(i-1)));
    S1_1a = S1_1a + dt / tau;
    % 公式1: 计算虚拟温度 
    Tf_1a(i) = Ta_1a + (T0 - Ta_1a) * exp(-(S1_1a)^beta);
end

%% 3. 图1b: 双步退火 - 高温至低温 (Monotonic)
T1_1b = 383; t1_total_1b = 50; % 预退火 [cite: 37, 272]
T2_1b = 373;                   % 第二步退火 [cite: 37, 272]
t2_1b = logspace(-1, 5, 1000); % 第二步时间范围 10^-1 到 10^5 s

% 预退火阶段 (Step 1)
t1_sim = linspace(0, t1_total_1b, 500);
Tf_step1 = T0;
S1_1b = 0;
for i = 2:length(t1_sim)
    dt = t1_sim(i) - t1_sim(i-1);
    tau = exp(log(A) + (x * H_star) / (R * T1_1b) + ((1 - x) * H_star) / (R * Tf_step1));
    S1_1b = S1_1b + dt / tau;
    Tf_step1 = T1_1b + (T0 - T1_1b) * exp(-(S1_1b)^beta);
end

% 第二步退火阶段 (Step 2)
Tf_1b = zeros(size(t2_1b));
Tf_1b(1) = Tf_step1;
S2_1b = 0;
for j = 2:length(t2_1b)
    dt = t2_1b(j) - t2_1b(j-1);
    tau = exp(log(A) + (x * H_star) / (R * T2_1b) + ((1 - x) * H_star) / (R * Tf_1b(j-1)));
    S2_1b = S2_1b + dt / tau;
    % 基于公式2的等效展开形式计算Tf 
    Tf_1b(j) = T2_1b + (T0 - T1_1b) * exp(-(S1_1b + S2_1b)^beta) + (T1_1b - T2_1b) * exp(-(S2_1b)^beta);
end

%% 4. 图1c: 双步退火 - 低温至高温 (Memory Effect)
T1_1c = 373; t1_total_1c = 50; % 预退火 [cite: 38, 274]
T2_1c = 383;                   % 第二步退火 [cite: 38, 274]
t2_1c = logspace(-1, 4, 1000); % 第二步时间范围 10^-1 到 10^4 s

% 预退火阶段 (Step 1)
t1_sim = linspace(0, t1_total_1c, 500);
Tf_step1 = T0;
S1_1c = 0;
for i = 2:length(t1_sim)
    dt = t1_sim(i) - t1_sim(i-1);
    tau = exp(log(A) + (x * H_star) / (R * T1_1c) + ((1 - x) * H_star) / (R * Tf_step1));
    S1_1c = S1_1c + dt / tau;
    Tf_step1 = T1_1c + (T0 - T1_1c) * exp(-(S1_1c)^beta);
end

% 第二步退火阶段 (Step 2)
Tf_1c = zeros(size(t2_1c));
Tf_1c(1) = Tf_step1;
S2_1c = 0;
for j = 2:length(t2_1c)
    dt = t2_1c(j) - t2_1c(j-1);
    tau = exp(log(A) + (x * H_star) / (R * T2_1c) + ((1 - x) * H_star) / (R * Tf_1c(j-1)));
    S2_1c = S2_1c + dt / tau;
    Tf_1c(j) = T2_1c + (T0 - T1_1c) * exp(-(S1_1c + S2_1c)^beta) + (T1_1c - T2_1c) * exp(-(S2_1c)^beta);
end

%% 5. 绘图展示
figure('Position', [100, 200, 1200, 350]);

% 图 1a
subplot(1,3,1);
semilogx(t1_1a, Tf_1a, 'b-', 'LineWidth', 2);
xlabel('t_1 (s)', 'FontSize', 12);
ylabel('Fictive Temperature T_f (K)', 'FontSize', 12);
title('(a) Monotonic (383 K)', 'FontSize', 12);
set(gca, 'YDir', 'reverse'); % 翻转Y轴以匹配原图焓下降的视觉趋势
grid on;

% 图 1b
subplot(1,3,2);
semilogx(t2_1b, Tf_1b, 'b-', 'LineWidth', 2);
xlabel('t_2 (s)', 'FontSize', 12);
ylabel('Fictive Temperature T_f (K)', 'FontSize', 12);
title('(b) Monotonic (383 K -> 373 K)', 'FontSize', 12);
set(gca, 'YDir', 'reverse');
grid on;

% 图 1c
subplot(1,3,3);
semilogx(t2_1c, Tf_1c, 'r-', 'LineWidth', 2);
xlabel('t_2 (s)', 'FontSize', 12);
ylabel('Fictive Temperature T_f (K)', 'FontSize', 12);
title('(c) Memory (373 K -> 383 K)', 'FontSize', 12);
set(gca, 'YDir', 'reverse');
grid on;
